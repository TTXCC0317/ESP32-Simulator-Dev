### 2.1 circuit.ts — 电路文档模型

```ts
export type PartId = string;   // nanoid，如 "p_7f3k..."
export type PinRef = `${PartId}:${string}`;  // "esp:GPIO4"

export interface PartInstance {
  id: PartId;
  type: string;                 // 元件类型，如 "wokwi-led"
  left: number; top: number;    // 画布坐标（px，网格 20px）

}

  color: WireColor;             // 8 色枚举

export interface CircuitDoc {
  formatVersion: 1;

export interface CircuitValidation {
```

### 2.2 catalog.ts — 元件/板卡定义

```ts
export interface PinDef {
  name: string;                 // 引脚名，如 "A"、"GPIO4"
  // N1 修正：role 枚举统一覆盖板卡与元件用例，避免 zod 拒
  //   板卡引脚用 'power'|'gnd'|'gpio'|'i2c.sda'|'i2c.scl'|'spi.mosi'|'spi.miso'|'spi.sck'|'spi.cs'|'uart.tx'|'uart.rx'|'pwm'|'adc'
  //   元件引脚用 'power'|'gnd'|'analog'|'signal.in'|'signal.out'|'signal.io'|'passive'
  // direction 仅板卡引脚用（'in'/'out'/'io'），元件 role 自带方向（signal.* / passive）
  role: 'power'|'gnd'|'gpio'|
        'i2c.sda'|'i2c.scl'|
        'spi.mosi'|'spi.miso'|'spi.sck'|'spi.cs'|
        'uart.tx'|'uart.rx'|
        'pwm'|'adc'|                    // 板卡引脚特殊功能
        'analog'|                       // 元件模拟信号（电位器 SIG）
        'signal.in'|'signal.out'|'signal.io'|   // 元件数字信号（带方向）
        'passive';                      // 元件无极性 passive（电阻两端）
  direction?: 'in'|'out'|'io';          // 仅板卡引脚用
  x: number; y: number;         // 相对元件左上角的引脚坐标（SVG 封装内）
}

export interface AttrDef {
  key: string;
  type: 'enum'|'number'|'color'|'boolean'|'text';
  label: string;
  default: string | number | boolean;
  options?: Array<{ value: string; label: string }>;
  min?: number; max?: number; step?: number;
}

export interface PartDefinition {
  type: string;
  name: string;
  category: 'mcu'|'io'|'sensor'|'display'|'power';
  defVersion: 1;
  pins: PinDef[];
  attrs: AttrDef[];
  renderer: { asset: string; width: number; height: number };  // assets/parts/<asset>
  simulator: {
    listens: EngineEventType[];   // 消费的引擎事件
    produces?: InputEventType[];  // 可注入的输入
    behavior: string;             // 见《05-元件清单》行为列；'i2c-device'/'spi-device' 触发查 device
    device?: DeviceSpec;           // M8：I2C/SPI 设备语义表（behavior 为 i2c-device/spi-device 时必填）
  };
}

/** M8 D3：I2C/SPI 设备寄存器语义，双端共同解释（引擎A PinBus / 引擎B gpioBridge） */
export interface I2cRegisterSpec {
  addr: number;          // 寄存器/命令字节 0x00–0xFF（7-bit 地址限制在 I2cDeviceSpec.address）
  size: number;          // 单次读取字节数
  decode?: 'raw'|'lux'|'be.int16'|'le.uint16'|'accel.xyz'|'gyro.xyz'|'whoami';
  defaultBytes?: number[];  // 无注入时默认字节（按 size 填充）
}
export interface I2cDeviceSpec { kind: 'i2c-device'; address: number; registers: I2cRegisterSpec[] }
export interface SpiDeviceSpec { kind: 'spi-device'; csGpio: number; registers?: I2cRegisterSpec[] }
export type DeviceSpec = I2cDeviceSpec | SpiDeviceSpec;

export interface BoardDefinition {
  type: string; name: string;
  mcu: 'esp32'|'esp32s3'|'esp32c3';
  arch: 'xtensa'|'riscv32';
  engines: Array<'micropython-wasm'|'qemu-remote'>;
  pins: Array<{ name: string; gpio: number; x: number; y: number;
                caps: Array<'gpio'|'pwm'|'adc'|'i2c.sda'|'i2c.scl'|'uart'|'spi'|'power'|'gnd'> }>;
}
```

### 2.3 engine.ts — 引擎抽象

```ts
export type EngineEventType =
  | 'gpio.write' | 'pwm.duty' | 'uart.rx' | 'i2c.txn' | 'spi.txn'
  | 'fb.update' | 'log' | 'state';

export type InputEventType =
  | 'pin.level' | 'analog.value' | 'sensor.data' | 'uart.tx' | 'key.event';

export interface EngineEventMap {
  'gpio.write': { pin: number; level: 0 | 1; seq: number };
  'pwm.duty':   { pin: number; duty: number; freq: number };
  'uart.rx':    { bytes: Uint8Array; port: 0 | 1 | 2 };
  'i2c.txn':    { addr: number; dir: 'r' | 'w'; data: Uint8Array; seq: number };
  'spi.txn':    { cs: number; data: Uint8Array; seq: number };
  'fb.update':  { partId: PartId; rect: [number, number, number, number]; data: Uint8Array };
  'log':        { level: 'info'|'warn'|'error'; text: string };
  'state':      { status: EngineStatus; error?: string };
}

export type EngineStatus = 'idle'|'loading'|'running'|'paused'|'error';

export interface FirmwareSourceFiles { kind: 'sources'; files: Array<{ path: string; content: string }> }
export interface FirmwareBinary { kind: 'binary'; flashImg: Uint8Array; entry?: number }  // 引擎B经后端；entry=app 分区偏移（默认 0x10000）
export type FirmwareInput = FirmwareSourceFiles | FirmwareBinary;

export interface SimulationEngine {
  readonly kind: 'micropython-wasm' | 'qemu-remote';
  load(circuit: CircuitDoc, fw: FirmwareInput): Promise<void>;   // N3: 引擎A 接 sources；引擎B 接 binary（由 BuildService 预编译后传入）
  start(opts?: { speed?: 0.25|0.5|1|2|4 }): void;
  pause(): void; reset(): void; dispose(): void;
  on<K extends EngineEventType>(type: K, cb: (p: EngineEventMap[K]) => void): () => void;
  input(ev: InputEvent): void;
}

// ---- N2: 三套状态枚举映射（避免前端实现无据） ----
// EngineStatus       引擎本地状态（§2.3，引擎A Worker 内部）
// ServerMsg.state    WS 协议状态（§2.4，引擎B 后端推送）
// simStore.status    前端展示状态（§6.2，统一为 EngineStatus + 'building'）
//
// 映射表（前端 simStore 收到 WS state 时按此转换）：
// ┌─────────────────┬──────────────────────┬──────────────────┐
// │ EngineStatus    │ ServerMsg.state      │ simStore.status  │
// ├─────────────────┼──────────────────────┼──────────────────┤
// │ idle            │ (无 WS，未 attach)    │ 'idle'           │
// │ loading         │ 'attaching'          │ 'loading'        │
// │ (无)            │ 'building-wait'      │ 'building'       │ ← 引擎B 编译等待，引擎A 不出现
// │ running         │ 'running'            │ 'running'        │
// │ paused          │ 'paused'            │ 'paused'          │
// │ error           │ 'error'             │ 'error'           │
// │ (无)            │ 'closed'            │ 'idle' (会话已关闭，回空闲) │
// └─────────────────┴──────────────────────┴──────────────────┘
//
// 引擎A 不出现 'building-wait'/'closed'（无后端编译、无会话概念）；
// 引擎B 不出现 'loading'（attach 时已是 building-wait，binary 由后端 BuildService 准备）；
// simStore 暴露给 UI 的状态码用 'idle'|'loading'|'building'|'running'|'paused'|'error' 6 个值。

export type InputEvent =
  | { type: 'pin.level';   partId: PartId; pin: string; level: 0|1 }
  | { type: 'analog.value';partId: PartId; pin: string; value: number }   // 0..4095
  | { type: 'sensor.data'; partId: PartId; data: Record<string, number> } // {temp: 25.5, hum: 60}
  | { type: 'uart.tx';     bytes: Uint8Array; port?: 0|1|2 };
```

### 2.4 ws-protocol.ts — WS 消息信封（引擎B）

```ts
export type ClientMsg =
  | { type: 'attach';     payload: { projectId: string; circuit: CircuitDoc; firmwareId: string; boardType: string } }

  | { type: 'build.progress'; payload: { buildId: string; phase: 'queued'|'compiling'|'linking'|'merging'|'success'|'failed'; progress: number; logLine?: string; logLines?: string[]; error?: string } }   // 编译进度推送（M4，详述见 §7.4 N19）
  | { type: 'gpio.write'; payload: { pin: number; level: 0|1; seq: number } }
  | { type: 'pwm.duty';   payload: { pin: number; duty: number; freq: number } }
  | { type: 'uart.rx';    payload: { bytes: number[] } }
  | { type: 'i2c.txn';    payload: { addr: number; dir: 'r'|'w'; data: number[]; seq: number } }
  | { type: 'spi.txn';    payload: { cs: number; data: number[]; seq: number } }
  | { type: 'sensor.data'; payload: { partId: PartId; data: Record<string, number>; gpio: number; seq: number } }   // 环境传感器读数推送（M8 DHT22 等 env-sensor，onDhtTxn 时推送）
  | { type: 'fb.update';  payload: { partId: PartId; rect: [number,number,number,number]; data: number[] } }
  | { type: 'log';        payload: { level: 'info'|'warn'|'error'; text: string } }
  | { type: 'error.ack';  payload: { code: string; message: string } }   // 非法消息回执
  | { type: 'pong';       payload: Record<string, never> };                // 心跳回应（ClientMsg.ping 的回执）
```

### 2.5 zod schema 样例（`packages/shared/src/ws-protocol.ts`）

> §2.4 注释说明 schema 与类型同文件维护；这里给出实际样例，作为 ws 网关前后端校验的单一来源。其他文件（circuit.ts/catalog.ts/engine.ts）按相同模式同文件维护 schema。

```ts
import { z } from 'zod';

// ---- ClientMsg ----
export const attachSchema = z.object({
  type: z.literal('attach'),
  payload: z.object({
    projectId: z.string().min(1),
    circuit: z.object({
      formatVersion: z.literal(1),
      boardType: z.string().min(1),
      parts: z.array(z.object({
        id: z.string().min(1),
        type: z.string().min(1),
        left: z.number().finite(),
        top: z.number().finite(),
        rotate: z.union([z.literal(0), z.literal(90), z.literal(180), z.literal(270)]),
        attrs: z.record(z.string(), z.union([z.string(), z.number(), z.boolean()])),
      })),
      connections: z.array(z.object({
        id: z.string().min(1),
        source: z.string().min(1),          // PinRef = `${PartId}:${pin}`
        target: z.string().min(1),
        color: z.enum(['black','red','green','blue','yellow','orange','purple','gray']),
        path: z.array(z.object({
          dir: z.enum(['v','h','*']),
          len: z.number().int(),
        })),
      })),
      serialMonitor: z.object({ baudrate: z.number().int().positive() }),
    }),
    firmwareId: z.string().min(1),
    boardType: z.string().min(1),
  }),
});

export const inputPinSchema = z.object({
  type: z.literal('input.pin'),
  payload: z.object({
    partId: z.string().min(1),
    pin: z.string().min(1),
    level: z.union([z.literal(0), z.literal(1)]),
    /** 按键松开（05-§1.4）：GPIO 桥解除注入，固件侧回退 pull 电平；level 被忽略 */
    release: z.boolean().optional(),
  }),
});

export const inputAnalogSchema = z.object({
  type: z.literal('input.analog'),
  payload: z.object({
    partId: z.string().min(1),
    pin: z.string().min(1),
    value: z.number().int().min(0).max(4095),     // ADC 12 位
  }),
});

export const inputUartSchema = z.object({
  type: z.literal('input.uart'),
  payload: z.object({ bytes: z.array(z.number().int().min(0).max(255)) }),
});

export const ctrlSchema = z.object({
  type: z.literal('ctrl'),
  payload: z.enum(['start','pause','reset','stop']),
});

export const clientMsgSchema = z.discriminatedUnion('type', [
  attachSchema, inputPinSchema, inputAnalogSchema, inputUartSchema, ctrlSchema,
]);
export type ClientMsg = z.infer<typeof clientMsgSchema>;   // 与 §2.4 手写类型对齐
```

**用法**：

```ts
// 后端 ws 网关
const parsed = clientMsgSchema.safeParse(JSON.parse(raw));
if (!parsed.success) {
  ws.send(JSON.stringify({ type: 'error.ack', payload: { code: 'WS_MSG_INVALID', message: parsed.error.message } }));
  return;
}
// 前端发送前自校验（避免发垃圾）
const raw = clientMsgSchema.parse(msg); ws.send(JSON.stringify(raw));
```

**与 §2.4 手写类型的关系**：zod `z.infer` 与手写 interface 必须对齐，CI 加一条快照测试（`expect<keyof ClientMsg>()` 全枚举覆盖），改类型/schema 必须同步。其余 ServerMsg、CircuitValidation、InputEvent 同模式补全 schema。

**M8 catalog device schema（`packages/shared/src/catalog.ts`，与 §2.2 类型同文件维护）**：I2C/SPI 设备寄存器语义表，双端（引擎A PinBus / 引擎B gpioBridge）共同解释。

```ts
export const i2cRegisterSpecSchema = z.object({
  addr: z.number().int().min(0).max(0xff),        // 寄存器/命令字节；7-bit 地址限制在 I2cDeviceSpec.address
  size: z.number().int().min(0).max(255),         // 单次读取字节数
  decode: z.enum(['raw','lux','be.int16','le.uint16','accel.xyz','gyro.xyz','whoami']).optional(),
  defaultBytes: z.array(z.number().int().min(0).max(255)).optional(),
});
export const i2cDeviceSpecSchema = z.object({
  kind: z.literal('i2c-device'),
  address: z.number().int().min(0).max(0x7f),     // 7-bit I2C 地址（BH1750=0x23, MPU6050=0x68）
  registers: z.array(i2cRegisterSpecSchema),
});
export const spiDeviceSpecSchema = z.object({
  kind: z.literal('spi-device'),
  csGpio: z.number().int().nonnegative(),         // CS 片选对应板卡 GPIO
  registers: z.array(i2cRegisterSpecSchema).optional(),
});
export const deviceSpecSchema = z.discriminatedUnion('kind', [i2cDeviceSpecSchema, spiDeviceSpecSchema]);
// partDefinitionSchema.simulator.device: deviceSpecSchema.optional()
```

**用法（双端共同解释）**：元件 JSON 声明 `simulator.behavior='i2c-device'` + `simulator.device={kind,address,registers}`；引擎A PinBus `Map<addr, VirtualDevice>` 同步应答 `readfrom`；引擎B gpioBridge 收到 I2C_TXN 帧按 `address` 查表 → 寄存器 `decode` 语义生成 SENSOR_REPLY 回帧。无设备挂载的 addr/cs 宿主回 NACK 语义（len=0）。

### 2.6 worker-protocol.ts — 引擎A Worker 消息（N7）

> §3.5 EngineWorker 骨架 `import type { WorkerMsg, WorkerReply } from '@esp32-sim/shared'`，本节给出实际类型 + zod schema，与 §2.3 EngineEventMap / EngineStatus 对齐。

```ts
import { z } from 'zod';
import type { CircuitDoc, FirmwareInput, EngineEventType, EngineEventMap, EngineStatus, InputEvent } from './engine';
import type { PartId } from './circuit';

// ---- WorkerMsg（主线程 → Worker） ----
export type WorkerMsg =
  | { type: 'load';    payload: { circuit: CircuitDoc; fw: FirmwareInput } }
  | { type: 'start';   payload: { speed?: 0.25|0.5|1|2|4 } }
  | { type: 'pause' }
  | { type: 'reset' }
  | { type: 'input';   payload: InputEvent }
  | { type: 'dispose' };

// ---- WorkerReply（Worker → 主线程） ----
export type WorkerReply =
  | { type: 'ready';       payload: { wasmMemBytes: number } }
  | { type: 'event.batch'; payload: { events: EngineEvent[] } }      // 16ms 批量聚合
  | { type: 'state';       payload: { status: EngineStatus; error?: string } }
  | { type: 'log';         payload: { level: 'info'|'warn'|'error'; text: string } };

// 引擎事件统一 envelope（§3.5 worker 骨架用）
export type EngineEvent =
  | { kind: 'gpio.write'; pin: number; level: 0|1; seq: number }
  | { kind: 'pwm.duty';   pin: number; duty: number; freq: number }
  | { kind: 'uart.rx';    bytes: Uint8Array; port: 0|1|2 }
  | { kind: 'i2c.txn';    addr: number; dir: 'r'|'w'; data: Uint8Array; seq: number }
  | { kind: 'spi.txn';    cs: number; data: Uint8Array; seq: number }
  | { kind: 'fb.update';  partId: PartId; rect: [number,number,number,number]; data: Uint8Array }
  | { kind: 'log';        level: 'info'|'warn'|'error'; text: string }
  | { kind: 'state';      status: EngineStatus; error?: string };

// ---- zod schema ----
export const workerMsgSchema = z.discriminatedUnion('type', [
  z.object({ type: z.literal('load'),    payload: z.object({
    circuit: z.unknown(),  // CircuitDoc schema 见 circuit.ts，转交校验
    fw: z.union([
      z.object({ kind: z.literal('sources'), files: z.array(z.object({ path: z.string(), content: z.string() })) }),
      z.object({ kind: z.literal('binary'),  flashImg: z.instanceof(Uint8Array), entry: z.number().optional() }),
    ]),
  }) }),
  z.object({ type: z.literal('start'),   payload: z.object({ speed: z.union([z.literal(0.25),z.literal(0.5),z.literal(1),z.literal(2),z.literal(4)]).optional() }) }),
  z.object({ type: z.literal('pause') }),
  z.object({ type: z.literal('reset') }),
  z.object({ type: z.literal('input'),  payload: z.unknown() }),  // InputEvent schema 见 engine.ts
  z.object({ type: z.literal('dispose') }),
]);
```

> **校验策略**：Worker 内 `self.onmessage` 入口先 `workerMsgSchema.safeParse`，失败 `postMessage({type:'log', payload:{level:'error', text:'bad msg'}})` 并跳过；不引入 `error.ack`（仅 WS 协议有）。`Uint8Array` 经结构化克隆天然支持，不需要 base64 转换。
>
> **`packages/shared/src/index.ts` 导出**：`export * from './worker-protocol'`，前端 worker 与 client 都从 `@esp32-sim/shared` 导入。

### 3.1 类结构

```
├────────────────────────────────────────────┤
```

### 3.2 machine shim → PinBus 转发表（M7 更新：PWM/ADC 已交付）

| MicroPython API | shim 行为 |
|---|---|
| `Pin(n, Pin.OUT)` | 注册输出驱动 `bus.claimOutput(n)` |
| `Pin.value(v)` | `bus.write(n, v)` → 触发 `gpio.write` 事件 |
| `Pin(n, Pin.IN, pull)` | `bus.claimInput(n, pull)` |
| `Pin.value()` | `bus.read(n)`（网络聚合电平） |
| `Pin.irq(fn, trigger)` | 订阅 `bus.onChange(n)` |
| `UART(0, baud)` | `bus.uartOpen(0, baud)`；`write()` → `uart.rx` 事件；`read()` ← `uart.tx` 输入 |
| **`PWM(pin, freq=1000, duty=0)`** · `pwm.duty([n])` · `pwm.freq([n])` | **`bus.pwm(n, duty, freq)` → 触发 `pwm.duty` 事件；duty 0–1023（10 位），freq Hz。接受 int 或 Pin 对象作引脚（M7 交付）** |
| **`ADC(Pin(n))` 或 `ADC(pin)` · `adc.read()` · `adc.read_u16()`** | **`bus.adcRead(n)` 同步返回 12 位 0–4095（无注入返回 0）；`read_u16()` = read()×16 线性映射到 0–65520（M7 交付）** |
| `I2C(0, scl, sda)` / `SoftI2C` | `bus.i2cController(scl, sda)`；`readfrom/writeto` → `i2c.txn` 事件（M8） |
| `SPI(baud)` | 同上，`spi.txn`（M8） |
| `neopixel` / RMT | 逐像素 → `fb.update`/专用事件（M9） |

### 3.3 关键机制
- **停止中断**：`pause/dispose` 通过 wasm 导出的 `mp_sched_keyboard_interrupt()` 注入 KeyboardInterrupt，死循环可中断（E1-04；v1.26 API，旧名 `mp_keyboard_interrupt` 已废弃）；
- **输入注入（M5）**：`gpioConfigure(pin, mode)` 记录 pull 到注入表（INPUT_PULLUP→1 / INPUT_PULLDOWN→0）；按键/开关注入（05-§1.4）→ `mp_js_gpio_inject(pin, level)` 沿触发已注册的 `Pin.irq`，注入表优先于网络电平供 `Pin.value()` 读取；release 清除注入并回退 pull 电平（机器 shim `apps/web/src/sim/mpy/machine-shim/machine.c`，引擎B对齐实现见 §7.2.2）；
- **stdout 捕获**：wasm `stdout_write` 回调 → `uart.rx`（port=0 REPL）+ 串口终端显示；
- **时间源**：`time.sleep` 经 mphalport 补丁映射到 `emscripten_sleep`（Asyncify 让出事件循环，不阻塞 JS 主线程/Worker）；`speed` 倍率缩放引擎A M4 不生效（wasm 内 sleep 无法缩放，M5 处理，实现见 `apps/web/src/sim/mpy/engine.ts`）；
- **VFS**：load() 将 files 写入内存文件系统（`/main.py` 等），支持 import 相对模块。

### 3.5 EngineWorker 消息协议（C1 引擎A Worker 通信协议）

> §9 提及 M4 起 EngineWorker 化。Worker 与主线程通过 `postMessage` 双向通信，消息为 tagged union，所有 payload 经 `packages/shared` 的 zod schema 校验。

**主线程 → Worker**：

| type | payload | 说明 |
|---|---|---|
| `load` | `{ circuit: CircuitDoc; files: FirmwareSourceFiles['files'] }` | 装载电路与固件源 |
| `start` | `{ speed?: number }` | 启动执行 |
| `pause` | — | 注入 KeyboardInterrupt |
| `reset` | — | 重置 wasm 实例（保留 VFS） |
| `input` | `InputEvent`（§2.3） | 元件输入注入 |
| `dispose` | — | 销毁 wasm，worker 自 terminate |

**Worker → 主线程**：

| type | payload | 说明 |
|---|---|---|
| `event` | `EngineEventMap[K]` + `{ kind: EngineEventType }` | 引擎输出事件（gpio.write/uart.rx/…），按 §2.3 EngineEventMap 序列化 |
| `state` | `{ status: EngineStatus; error?: string }` | 状态变化 |
| `log` | `{ level: 'info'\|'warn'\|'error'; text: string }` | 引擎内部日志 |
| `ready` | `{ wasmMemBytes: number }` | load 完成，回传实际占用内存（用于 §06 边界 256MB 上限监控） |

**序列化约束**：Uint8Array 用 `postMessage` 自动结构化克隆传输（不 Transfer，保留 worker 内副本）；高频 `gpio.write`/`uart.rx` 批量发送（16ms 窗口聚合为 `event.batch`，payload `{events: EngineEvent[]}`），主线程 simStore 收 `event.batch` 后逐项分发，rAF flush。

> **N22 决策落地**：本协议**不依赖 SharedArrayBuffer**——所有事件走 `postMessage` 结构化克隆 + 16ms 批量聚合；高频 GPIO 写入（1kHz+）有 16ms 延迟，串口高带宽走批量字节（每批 ≤4KB）规避序列化压力。COOP/COEP 头不再需要，部署链路简化。

**Worker 文件骨架**（`apps/web/src/sim/mpy/engine.worker.ts`）：

```ts
/// <reference lib="webworker" />
import type { WorkerMsg, WorkerReply } from '@esp32-sim/shared';
import { MpyWasmEngine } from './engine';

const engine = new MpyWasmEngine();
const batch: EngineEvent[] = [];           // 16ms 窗口聚合缓冲
let flushScheduled = false;

function scheduleFlush() {
  if (flushScheduled) return;
  flushScheduled = true;
  setTimeout(() => {
    if (batch.length) postMessage({ type: 'event.batch', payload: { events: batch.splice(0) } } satisfies WorkerReply);
    flushScheduled = false;
  }, 16);
}

self.onmessage = async (e: MessageEvent<WorkerMsg>) => {
  const msg = e.data;
  switch (msg.type) {
    case 'load':
      await engine.load(msg.payload.circuit, msg.payload.files);
      postMessage({ type: 'ready', payload: { wasmMemBytes: engine.memBytes() } } satisfies WorkerReply);
      break;
    case 'start': engine.start(msg.payload); break;
    case 'pause': engine.pause(); break;
    case 'reset': engine.reset(); break;
    case 'input': engine.input(msg.payload); break;
    case 'dispose': engine.dispose(); self.close(); break;
  }
};
// engine.on 事件聚合到 batch（不直接 postMessage，避免高频扇出）
engine.on('gpio.write', p => { batch.push({ kind: 'gpio.write', ...p }); scheduleFlush(); });
engine.on('uart.rx',  p => { batch.push({ kind: 'uart.rx',  ...p }); scheduleFlush(); });
engine.on('state',    p => postMessage({ type: 'state', payload: p } satisfies WorkerReply));  // 状态立即送
engine.on('log',      p => postMessage({ type: 'log', payload: p } satisfies WorkerReply));
```

**主线程消费侧**（`apps/web/src/sim/mpy/engine-client.ts`，M4 实现）：

```ts
import { useSimStore } from '@/stores/sim';
import type { WorkerMsg, WorkerReply } from '@esp32-sim/shared';

export class EngineWorkerClient implements SimulationEngine {
  private worker: Worker;
  private pendingEvents: EngineEvent[] = [];      // rAF 合并缓冲（与 worker 侧 batch 二次合并）
  private rafId = 0;
  readonly kind = 'micropython-wasm' as const;

  constructor() {
    this.worker = new Worker(new URL('./engine.worker.ts', import.meta.url), { type: 'module' });
    this.worker.onmessage = (e: MessageEvent<WorkerReply>) => this.onReply(e.data);
  }

  private onReply(msg: WorkerReply) {
    switch (msg.type) {
      case 'event.batch':
        this.pendingEvents.push(...msg.payload.events);
        if (!this.rafId) this.rafId = requestAnimationFrame(this.flush);
        break;
      case 'state':    useSimStore.getState().setStatus(msg.payload.status, msg.payload.error); break;
      case 'log':      useSimStore.getState().appendLog(msg.payload); break;
      case 'ready':    useSimStore.getState().setWasmMemBytes(msg.payload.wasmMemBytes); break;
    }
  }

  private flush = () => {
    const events = this.pendingEvents.splice(0);
    this.rafId = 0;
    // 分发：simStore.attach(engine) 时订阅的回调逐个调用
    for (const ev of events) this.subscribers[ev.kind]?.(ev);
  };

  // SimulationEngine 接口实现：转发到 worker
  async load(c, fw) { this.post({ type: 'load', payload: { circuit: c, files: fw.files } }); await onceReady(); }
  start(opts?) { this.post({ type: 'start', payload: opts ?? {} }); }
  pause()      { this.post({ type: 'pause' }); }
  reset()      { this.post({ type: 'reset' }); }
  input(ev)    { this.post({ type: 'input', payload: ev }); }
  dispose()    { this.post({ type: 'dispose' }); this.worker.terminate(); }

  private post(m: WorkerMsg) { this.worker.postMessage(m); }
  private subscribers: { [K in EngineEventType]?: (p: EngineEventMap[K]) => void } = {};
  on<K extends EngineEventType>(type: K, cb: (p: EngineEventMap[K]) => void) {
    this.subscribers[type] = cb as any;
    return () => { if (this.subscribers[type] === cb) delete this.subscribers[type]; };
  }
}
```

> 主线程二次合并：worker 已 16ms 批量，主线程 rAF 再合并多批到一帧（约 16ms）刷新元件视图，避免每个 `gpio.write` 都触发 Konva 重绘。串口终端走 `xterm.js` 的 `write()` 累积写入（不阻塞 rAF）。

### 4.1 数据模型

```ts
class PinBus {
  // 网络（net）= 电气相连引脚的集合（由 connections 聚合，Union-Find）
  private nets: Map<NetId, {
    members: Array<{ partId: PartId; pin: string }>;
    drivers: Map<DriverToken, 0|1>;      // 输出驱动（板卡GPIO/元件）
    inputReaders: Set<ReaderToken>;      // 输入读者（GPIO输入/元件）
    pull?: 'up'|'down'|'none';           // 板卡上拉/下拉
    analog: Map<ReaderToken, number>;    // 模拟注入值
    floating: boolean;                   // 无驱动 → 读默认电平
  }>;

  claimOutput(pinRef: PinRef): DriverToken;
  claimInput(pinRef: PinRef, pull: 'up'|'down'|'none'): ReaderToken;
  write(token: DriverToken, level: 0|1): void;   // → onChange 广播
  read(pinRef: PinRef): 0|1;                     // 无驱动: pull 决定，否则 0
  onChange(pinRef: PinRef, cb: (level: 0|1) => void): () => void;
  subscribe(pinRef: PinRef, cb: (event: PinBusEvent) => void): () => void;  // 06-§3.1 逻辑分析仪用：订阅该引脚的全部 PinBus 事件流（电平翻转、pwm.duty、i2c.txn 等），返回取消订阅函数；与 onChange 区别：onChange 仅电平变化时回调，subscribe 涵盖所有事件类型用于波形捕获
  conflict(n: NetId): boolean;                   // 多驱动电平不一致 → 冲突
  injectAnalog(pinRef: PinRef, value: number): void;

  // ---- N8 补全：§3.2 shim 转发表引用的所有方法 ----
  injectPin(pinRef: PinRef, level: 0|1): void;          // §8.3 按键元件用：非驱动源注入电平（与 write 区别：不 claim，瞬时）
  adcRead(pinRef: PinRef): number;                       // §3.2 ADC：读模拟值（0..4095），无注入返回 0（见 05 §1.6 E3）
  uartOpen(port: 0|1|2, baud: number): void;             // §3.2 UART：打开串口（占位，实际无硬件状态）
  pwm(pinRef: PinRef, duty: number, freq: number): void; // §3.2 PWM：触发 pwm.duty 事件
  i2cController(scl: PinRef, sda: PinRef): I2cController; // §3.2 I2C：返回控制器对象
  spiController(sck: PinRef, mosi?: PinRef, miso?: PinRef, cs?: PinRef[]): SpiController; // §3.2 SPI
}

// PinBus 引脚级事件流（subscribe 回调参数；06-§3.1 逻辑分析仪消费）
// 与 §2.3 EngineEventMap 区别：EngineEvent 是引擎 → 主线程的批量事件（经 worker 16ms 聚合）；
// PinBusEvent 是 PinBus 内部直接广播（主线程订阅，不经 worker），用于波形捕获的实时性
export type PinBusEvent =
  | { kind: 'pin.level';    pinRef: PinRef; level: 0|1; ts: number }
  | { kind: 'pwm.duty';     pinRef: PinRef; duty: number; freq: number; ts: number }
  | { kind: 'analog.value'; pinRef: PinRef; value: number; ts: number }  // 0..4095
  | { kind: 'i2c.txn';      pinRef: PinRef; addr: number; dir: 'r'|'w'; data: Uint8Array; ts: number }
  | { kind: 'spi.txn';      pinRef: PinRef; cs: number; data: Uint8Array; ts: number }
  | { kind: 'uart.tx';      pinRef: PinRef; bytes: Uint8Array; port: 0|1|2; ts: number }
  | { kind: 'uart.rx';      pinRef: PinRef; bytes: Uint8Array; port: 0|1|2; ts: number };

// I2C/SPI 控制器接口（PinBus 工厂返回）
interface I2cController {
  writeto(addr: number, data: Uint8Array): void;        // 触发 i2c.txn(dir='w')
  readfrom(addr: number, len: number): Uint8Array;      // 触发 i2c.txn(dir='r')，返回注入数据
}

interface SpiController {
  write(cs: number, data: Uint8Array): void;            // 触发 spi.txn
  read(cs: number, len: number): Uint8Array;
}
```

### 4.2 规则
1. **网络聚合**：load() 时对 connections 做 Union-Find，同网络内所有引脚同电平（逻辑拓扑，不做电阻分压，见《06-边界说明》§2）；
2. **多驱动冲突**：同网络两个输出驱动电平不一致时，取最后写入者并广播 `log.warn`（S3 级体验，不阻断）；
3. **电源语义**：VCC/GND 为固定网络（1/0）；LED 阴极接 GND 网络才点亮——元件 behavior 中声明引脚角色（见《05-元件清单》）；
4. **事件去重**：相同电平重复写不广播；高频事件由 analyzer 侧环形缓冲限流，PinBus 本身不丢事件。

#### 4.2.1 Union-Find 网络聚合伪代码（N9）

> §4.2 规则 1 + 05-§1.1.1 板卡同名引脚聚合都依赖本算法。给出伪代码避免实现分歧。

```
load(circuit: CircuitDoc):
  uf = new UnionFind()

  # 1. 板卡同名引脚预聚合（见 05 §1.1.1）
  #    BoardDefinition.pins 列出每个 GPIO 在左右列的位置（同 gpio 编号 → 同 PinRef）
  #    PinRef 命名 = `${boardId}:${gpioName}`，不带列后缀
  for pin in boardDef.pins:
    if pin.gpio is defined:                          # 仅 GPIO 引脚聚合（电源/地不聚合）
      pinRef = f"{boardId}:{pin.gpio}"               # 例 "esp:GPIO4"
      uf.makeSet(pinRef)                              # 多次 makeSet 同 key 是幂等的

  # 2. 把所有 PartInstance 的所有 PinDef 注册到 uf
  for part in circuit.parts:
    def = catalog.lookup(part.type)
    for pin in def.pins:
      pinRef = f"{part.id}:{pin.name}"                # 例 "led1:A"
      uf.makeSet(pinRef)

  # 3. 按 connections 合并（source 和 target 同网络）
  for conn in circuit.connections:
    uf.union(conn.source, conn.target)               # 例 "esp:GPIO4" ∪ "led1:A"

  # 4. 提取 NetId → 引脚集合
  nets: Map<NetId, Array<{partId, pin}>> = {}
  for pinRef in uf.allKeys():
    netId = uf.find(pinRef)
    nets[netId].push(parsePinRef(pinRef))

  # 5. 为每个 NetId 构建内部状态（drivers/inputReaders/pull/analog/floating）
  for netId, members in nets:
    nets[netId] = { members, drivers: new Map(), inputReaders: new Set(),
                    pull: 'none', analog: new Map(), floating: true }
```

**关键点**：
- `makeSet` 幂等：板卡左右列同名引脚（`esp:GPIO4`）只产生一个 PinRef，不会分裂为两个；
- `union` 把所有 connection 的两端并入同一集合；
- 多 driver 冲突（同 NetId 有两个 DriverToken 写不同电平）在 `write()` 时按规则 2 处理；
- 复杂度：O(n α(n))，n = 引脚总数（< 1000），α 是反 Ackermann，实测 < 1ms。

```
class UnionFind:
  parent: Map<PinRef, PinRef>
  makeSet(x): parent[x] = x
  find(x): if parent[x] != x: parent[x] = find(parent[x]); return parent[x]  # 路径压缩
  union(x, y): rx, ry = find(x), find(y); if rx != ry: parent[rx] = ry
```

### 6.4 前端本地缓存策略（IndexedDB）

> 后端 SQLite 为唯一权威数据源（§01 数据库设计）；前端是否要 IndexedDB 缓存？**需要，但仅作离线编辑/瞬时回放的辅助**，不做主存。

**场景**：
- **离线编辑**：用户打开工程后断网/后端临时不可达，仍可在画布继续拖元件/连线；恢复后批量 PUT 同步（带 `If-Match` 版本号防覆盖）。
- **运行态快照**：引擎A 运行中刷新页面会丢失运行状态，IndexedDB 暂存最近一次 `circuit + 引擎实例快照`（仅限引擎A，引擎B 状态在后端）。
- **大文件**：`main.py`/`sketch.ino` 超过 64KB 时，IndexedDB 存原始内容比 zustand 内存字符串更省内存（按需读入编辑器）。

**实现**：`apps/web/src/stores/idb.ts` 封装 [idb-keyval](https://github.com/jakearchibald/idb-keyval)（5KB，Promise API）。

| key | 值 | TTL |
|---|---|---|
| `project:<id>:draft` | `{ circuit, files, dirtyAt }` | 后端 PUT 成功后清除；超过 7 天清理 |
| `project:<id>:runtime` | `{ engineKind, wasmSnapshotBytes? }` | 引擎A dispose 时写；页面加载时若存在则提示"恢复上次运行" |
| `ui:layout` | 面板尺寸/折叠状态 | 永久（与 settings 表双写，前端优先读 IndexedDB 减少请求） |

**冲突与一致性**：
- 后端恢复后同步：前端先 GET 工程元数据 `updated_at`，若 IndexedDB `draft.dirtyAt` > 后端 `updated_at`，提示用户"本地有未保存改动，是否覆盖/合并/丢弃"；
- 不引入 PouchDB/CouchDB 同步：单机定位，自动合并复杂度不值得；用人工决策替代自动合并。

**边界**（与《06-边界说明》§4 对齐）：
- IndexedDB 单工程占用 ≤10MB（超出 `log.warn` 并丢弃最旧 runtime 快照）；
- 浏览器存储压力下 IndexedDB 可能被清理（隐私模式），前端检测失败时降级为"无缓存，必须在线编辑"，不阻断主流程。

#### 7.2.1 QEMU 启动参数（N17）

> M5 引擎B spawn 时按板卡 arch 选 qemu-system-xtensa（ESP32/S3）或 qemu-system-riscv32（C3）；参数数组化传给 execa（禁止 shell 拼接，见《06-边界说明》§6）。

**ESP32 / ESP32-S3（Espressif fork esp-develop 9.x，xtensa softmmu）**：

```ts
const args = [
  '-machine', boardType === 'esp32s3' ? 'esp32s3' : 'esp32',
  '-drive', `file=${flashImgPath},if=mtd,format=raw`,  // format=raw：消除裸镜像探测警告与 block0 写限制
  '-serial', `tcp:127.0.0.1:${port},server,nowait`,      // UART0：串口监视器
  '-serial', `tcp:127.0.0.1:${gpioPort},server,nowait`,  // UART1：GPIO 桥通道（M5，§7.2.2）
  '-monitor', 'none',
  '-nographic',                          // 不开图形窗口
  '-m', '4',                             // 4MB RAM（ESP32 实际 520KB SRAM + PSRAM 模拟）
  '-cpu', 'esp32',                       // 显式 CPU 型号
];
```

**ESP32-C3（riscv32）**：用 `qemu-system-riscv32 -machine esp32c3`，其余参数同上但 `-cpu 'riscv32'`。

**参数说明**：
- 不用 `-S`（启动暂停）：attach 后立即运行（用户已点 ▶）；若需 attach-then-run 改走 `monitor` 的 `cont` 命令；
- `-serial tcp:...,server,nowait`：QEMU 作为 TCP server，后端连上即开始双向数据流；
- 第二 `-serial`（M5 起）：UART1 作为 GPIO 桥通道——固件侧 glue 随 sketch 编译（tools/bridge-glue/esp32sim_bridge.c，强符号覆盖 esp32 core 3.x 的 GPIO HAL weak alias：`digitalWrite`/`pinMode` 等公开符号由 glue 强定义接管、转发 `__*` 真实实现并经 UART1 交换 5 字节帧；不用 lcgamboa fork 的 `-gpio-extract`（Espressif fork 不支持）；
- `-monitor none`：不暴露 QEMU monitor（避免用户随意 `qmp` 指令）；
- `-nographic`：禁用图形窗口（无头模式）；
- `-m 4`：4MB 内存足够 ESP32；PSRAM 场景加 `-m 8`；
- 不用 `-S`/`-loadvm` 等快照参数：单机定位，状态保存走 IndexedDB（引擎A）/会话级（引擎B），不做 QEMU 内部快照。

#### 7.2.2 QemuGpioBridge（M5 GPIO 桥·HAL 方案）

UART1 字节流 ⇄ **6 字节定长帧**（M7 升级；原 5 字节因 ADC 值 0–4095 超 1 字节、PWM duty 10 位同样超，统一 16 位 value = vH:vL）。tools/bridge-glue/esp32sim_bridge.c 与服务端 `gpio.bridge.ts` 双端严格同构；**M8 起新增 TLV 变长帧承载 I2C/SPI 事务（见下方"M8 TLV 变长帧"），状态机按 type 首字节分支解析**：

```
0xA5 | type | pin | vH | vL | xor(前5字节异或)     xor = A5 ^ type ^ pin ^ vH ^ vL
```

完整类型枚举（M5 保持 16 位兼容 vH=0，M7 新增 4 类）：

| type | 方向 | 含义 | vH:vL = uint16 value |
|---|---|---|---|
| 0x01 GPIO_WRITE | 固件→宿主 | 数字写 | vH=0，vL=0/1（level） |
| 0x02 PIN_MODE | 固件→宿主 | pinMode | vH=0，vL=Arduino mode 原字节（INPUT/OUTPUT/PULLUP/PULLDOWN 位） |
| **0x03 PWM_WRITE** | 固件→宿主 | **PWM duty（M7）** | value = duty 0–1023（LEDC 10 位归一化，WS 协议统一尺度） |
| **0x04 PWM_FREQ** | 固件→宿主 | **PWM 频率（M7）** | value = Hz（0–65535；首次 analogWrite 自动补发 1000 默认） |
| 0x11 GPIO_INPUT | 宿主→固件 | 数字注入 | vH=0，vL=0/1；glue 更新 s_inject 表 + 沿匹配触发 attachInterrupt 回调 |
| **0x12 ADC_INPUT** | 宿主→固件 | **模拟注入（M7）** | value = 0–4095（12 位钳位），glue analogRead 有注入返回注入值，无注入短路返回 0（05-§1.6 E3） |

**M8 TLV 变长帧**：6B 定长帧无法承载 I2C/SPI 事务（addr + len + data[] 变长载荷）。type ≥ 0x20 改用变长格式，帧状态机按 type 首字节分支（type ≤ 0x12 维持上表定长解析，二进制兼容 M5–M7 固件）：

```
0xA5 | type | len | [payload × len] | xor(前 len+3 字节异或)     len ≤ 255（06-§3 边界）
```

| type | 方向 | 含义 | payload |
|---|---|---|---|
| **0x20 I2C_TXN** | 固件→宿主 | **I2C 事务（M8）** | addr(7bit) | dir(1=读/0=写) | len | data[]；写事务 dir=0 带写数据，读事务 dir=1 data 为空、len 为请求字节数 |
| **0x21 SPI_TXN** | 固件→宿主 | **SPI 全双工事务（M8）** | cs_pin | len | data[]（发送内容，宿主按虚拟设备寄存器语义生成等长应答） |
| **0x22 DHT22_TXN** | 固件→宿主 | **DHT22 单总线请求（M8 后续）** | pin(1)——触发宿主查 dhtDevices 表回复温湿度 |
| **0x30 SENSOR_REPLY** | 宿主→固件 | **I2C 读应答（M8）** | addr | len | data[]（虚拟设备寄存器语义应答；无设备挂载时宿主回 NACK 语义 len=0） |
| **0x31 SPI_REPLY** | 宿主→固件 | **SPI 应答（M8）** | cs_pin | len | data[]（与 SPI_TXN 等长） |
| **0x32 DHT22_REPLY** | 宿主→固件 | **DHT22 应答（M8 后续）** | pin(1) | tempRaw_hi(1) | tempRaw_lo(1) | humRaw_hi(1) | humRaw_lo(1)；tempRaw/humRaw = Math.round(value × 10)（最高位=符号），无设备时回 0 防超时死锁 |

UART1 单通道内请求-应答闭环：glue 发出 I2C_TXN/SPI_TXN/DHT22_TXN 后同步等待对应 REPLY（忙等 + 50ms 超时，超时按"无设备"返回 0 字节，防宿主缺失死锁）；宿主 gpioBridge 收到事务帧查虚拟设备表（D3）→ 语义应答回帧。DHT22 不走 I2C 总线（role=signal.io），独立 dhtDevices 表 `Map<gpio, {temp, humidity}>` 由 buildDeviceTables 从 circuit connections 解析 pin 映射 + attrs 覆盖 defaults。

Arduino mode 位（esp32 core 3.x）：INPUT=0x01 / OUTPUT=0x03 / PULLUP=0x04 / PULLDOWN=0x08；`pullOfMode()` 把 PULLUP/PULLDOWN 位映射为 release 回退电平（1/0）。

**glue 侧 pull 表（M7 前实测修复）**：QEMU 不模拟 GPIO 内部上/下拉，悬空输入经 `__digitalRead` 恒读 0（上拉按键幻象按下）。glue（esp32sim_bridge.c）在 `pinMode` 时按位测试 PULLUP/PULLDOWN 位维护 `s_pull[pin]` 表；`digitalRead` 无注入时按 `s_pull` 声明返回电平（1/0），无 pull 声明回落 `__digitalRead`。宿主侧 `pullOfMode()` release 回退语义不变（见下文网关接线）。

**glue 侧 PWM/ADC 覆盖（M7 实测订正）**：

- **analogWrite/analogWriteFrequency 在 esp32 core 3.3.10 是强定义**（`esp32-hal-ledc.c` 直接定义，非 weak alias；不暴露 `__analogWrite`/`__analogWriteFrequency` 强符号供转调）。与 M5 GPIO `digitalWrite` weak alias 模式不同。glue 同强定义覆盖同名符号会触发链接器 multiple-definition 报错，BuildService 编译命令必须注入 `--build-property compiler.c.elf.extra_flags=-Wl,--allow-multiple-definition`（见 §7.1）。glue 内部用 ESP-IDF ledc API 直接驱动外设，core 原实现代码路径不再生效。
- **QEMU Espressif fork 不模拟 ADC 外设**（M7 servo-pot 100% Cache error 根因）：转调 `__analogRead` 进入 `adc_oneshot_ll_get_event` 等待不存在的 ADC 完成事件 → "Cache disabled but cached memory region accessed" panic（Core 1 backtrace 落 `adc_oneshot_ll_get_event → __analogRead → analogRead → loop`）。与 06-§3 "QEMU 不模拟 GPIO 内部上/下拉" 同源，扩展为 "QEMU 不模拟 ADC"。glue `analogRead` 短路：有注入返回注入值（`s_analog_inject[pin]`），无注入返回 0（与 05-§1.6 E3 "无注入读 0"、PinBus.adcRead 行为一致），永不转调 `__analogRead`。extern 声明已移除。
- **addr2line 回溯定位法**：用 `xtensa-esp32-elf-addr2line.exe -f -C -i -e main.ino.elf <addr>...` 解析 panic backtrace 地址，定位到具体源文件行号（M7 servo-pot 诊断用 `data/probe-qemu-uart-dump.mts` 抓完整 panic dump + addr2line 解析 12 个 backtrace 地址定位根因）。

**glue 侧 Wire/SPI 覆盖（M8 探针实证，data/probe-m8-bus.mts 三变体取证）**：

- **QEMU 基线行为（wire-base / spi-base）**：QEMU Espressif fork 不模拟 I2C/SPI 外设语义，但与 M7 ADC panic 先例**不同**——外设寄存器写被静默忽略/FIFO 直通，**不 panic 而是静默失败**：`Wire.begin` "成功"，`endTransmission` 恒返回 4（other error），`requestFrom` 恒 0 字节；`SPI.transfer` 回显发送值（0x9F 进 0x9F 出，FIFO 直通幻象应答，非真实从机）。固件"幻象运行"，比 panic 更隐蔽 → glue 必须**完全短路** Wire/SPI 方法，永不进入 ESP-IDF 驱动路径（同 M7 analogRead 短路原则）。
- **C++ 类成员函数链接级覆盖（wire-shim 实证胜出）**：`TwoWire`/`SPIClass` 是 C++ 类——成员函数既非 C 弱别名（M5 GPIO 模式）也非无副作用的强定义重复（M7 analogWrite 模式），无 `__Wire_*` 符号可转调。覆盖机制：glue C++ 源文件随 sketch 编译（同 esp32sim_bridge.c 模式），同 mangled name 强定义 + sketch 目标文件先于 core 库链接 + BuildService 已注入的 `-Wl,--allow-multiple-definition`（§7.1）→ glue 版本胜出，**core 与 Wire 库内部引用一并被接管**。探针证据（data/probe-wire-shim.log）：sketch 内自定义 `TwoWire::begin` 的 `MARK:SHIM-BEGIN-CALLED` 出现于 UART0，`endTransmission`/`requestFrom` 走短路虚拟应答（ENDTX-ERR=0、REQ-N=2），全程无 panic。SPI 覆盖同机制（SPIClass 成员函数同为 Itanium ABI mangled name），实现期无需单独覆盖探针。
- **探针可复现**：`cd apps/server && pnpm exec tsx ../../data/probe-m8-bus.mts <wire-base|wire-shim|spi-base>`（编译 + QEMU 12s UART 捕获 + 自动判定；日志 `data/probe-*.log`）。ESP32-C3（riscv32）mangled name 同 ABI 通用，覆盖机制预期同源，阶段 3 实现时补验。

**网关接线（ws-gateway）**：
- 固件→宿主：GPIO_WRITE → WS `gpio.write` 事件（前端 runtimeStore 汇聚渲染）；PIN_MODE → 记录 `pinPull`（release 回退用）；**PWM_WRITE 组合 `pinFreq`（PWM_FREQ 帧维护，默认 1000）→ WS `pwm.duty { pin, duty, freq }`**；
- 宿主→固件：WS `input.pin` → 邻接表 BFS 解析元件引脚到板卡 GPIO（attach 时由 circuit 构建）→ `bridge.injectInput(gpio, level)`；`release: true` 忽略 level，回退 `pinPull` 记录的 pull 电平（无记录按 0，05-§1.4）；**WS `input.analog { partId, pin, value }` → BFS 查对应 GPIO（要求 caps 含 `adc`，否则 `ADC_PIN_INVALID`）→ `bridge.injectAnalog(gpio, value)`，0–4095 钳位**；
- 桥 socket 断开：`input.pin` / `input.analog` 返回 `error.ack(NO_GPIO_BRIDGE)`，会话保持 running。

**Golden runner（apps/server/src/golden/）**：剧本 `input` 输入序列按 `atMs` 调度注入（release 回退 PIN_MODE pull），GPIO_WRITE 帧计数断言 `expect.gpio`；引擎A（micropython.wasm）同源注入走 machine shim（gpioRead 注入表 + `mp_js_gpio_inject` 沿触发 irq，release 回退 `gpioConfigure` pull）。release 时 pull 尚未观测到（固件慢启动，PIN_MODE/configure 上报晚于 release 调度）则取该脚最近一次按下电平取反（上拉对地 0→1、下拉接 VCC 1→0，与 pull 语义一致；M6 golden 回归实测修复的启动竞态）。

### 7.3 会话状态机（WS 网关）

> N11 补全：增加 `building-wait` 状态（attach 后等编译完成），明确所有转移条件。

```
                              attach（带 firmwareId）
   [closed] ─────────────────────────────────────► [attaching]
       ▲                                                  │
       │                                                  │ build 不存在 → [building-wait]
       │                                                  │ build 已存在 ──► [running]
       │                                                  ▼
       │                                            [building-wait]
       │                                                  │ build success ──► [running]
       │                                                  │ build failed ───► [error]
       │                                                  │ build timeout ──► [error]
       │                                                  │
       │                                                  ▼
       │             ┌───────────── [running] ⇄ [paused] ◄─── ctrl start/pause
       │             │                       │
       │             │                       │ ctrl stop / dispose / error
       │             │                       ▼
       │             │                  [error]
       │             │                       │
       └─────────────┴──────── stop/dispose ◄┘
```

**转移条件**：
- `closed → attaching`：客户端 `attach` 消息到达，分配 sessionId；
- `attaching → building-wait`：BuildService 查到 firmwareId 无成功 build，触发新编译；
- `attaching → running`：firmwareId 已有成功 build，直接 spawn QEMU；
- `building-wait → running`：BuildService 推送 `build.success`（经 §7.4 编译进度通道），后端 spawn QEMU；
- `building-wait → error`：build failed/timeout；
- `running ⇄ paused`：客户端 `ctrl start/pause`；M4 引擎B 例外——QEMU 进程级 pause 在 Windows 无 SIGSTOP 语义，`ctrl pause` 返回 `error.ack(unsupported)`，状态保持 running；`ctrl reset` 由网关 dispose + 重新 spawn 实现（respawn）；
- `running/paused → error`：QEMU 进程非零退出、网络错误、固件 panic；
- `任意 → closed`：客户端 `ctrl stop` 或后端 `dispose`（异常清理/超时回收）。

非法转移（如 `paused` 中再次 `attach`）→ `error.ack`，状态不变。断线保留 60s 供重连（见《06-边界说明》§7.1）。

### 7.4 编译进度推送通道（N19）

> §7.1 BuildService.onLog callback 仅是后端内部接口；前端如何感知编译进度？经 N19 决策：复用引擎B 的 WS 会话通道推送，不开新端口。

**协议**：客户端 `attach` 后进入 `building-wait` 状态时，后端把编译进度通过同一 WS 连接推送：

```ts
// ServerMsg 扩展（§2.4 已含 state，这里补充 build.progress）
| { type: 'build.progress'; payload: {
      buildId: string;
      phase: 'queued'|'compiling'|'linking'|'merging'|'success'|'failed';
      progress: number;           // 0..1（按 arduino-cli 输出行数估算）
      logLine?: string;           // arduino-cli stdout/stderr 当前行
      error?: string;             // failed 时的错误摘要
    }
}
```

**前端处理**：
- 收到 `build.progress` 时，UI 在底部面板 Logs Tab 显示进度条 + 行式日志；
- `phase: 'success'` 后立即收到 `state: running`（后端 spawn QEMU 完成），进度条消失；
- `phase: 'failed'` 后收到 `state: error`，进度条转红 + 显示 error 详情；
- 不轮询 REST：编译是异步过程，轮询既浪费又延迟；WS 推送实时性最佳；
- 引擎A 不出现 `build.progress`（无后端编译）。

**背压**：arduino-cli 输出可能很碎（每行一条 `build.progress`），后端在 WS 网关侧用 100ms 窗口聚合多行成一条推送（payload.logLines: string[]），避免前端被刷屏；critical 行（error/warning）立即推送不聚合。

## 9. 并发与线程模型

| 上下文 | 模型 | 约束 |
|---|---|---|
| 引擎A 执行 | WASM 在主线程 → 阻塞 UI；**M4 起 Worker 化**：EngineWorker（纯 `postMessage` + 16ms 批量聚合，**不依赖 SharedArrayBuffer/COOP/COEP**，详见 §3.5） | UI 帧率不受固件影响为硬性要求 |
| PinBus/渲染 | 主线程；事件经 Worker postMessage 批量到达，rAF 合并刷新 | 批处理窗口 16ms |
| 后端编译/QEMU | Node 主进程 + execa 子进程；WS 网关纯 IO | 编译日志行式转发，背压忽略（环形缓冲） |
